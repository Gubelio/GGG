country_event = {
    id = 50115
    title = "EVT50115NAME"
    desc = "EVT50115DESC"
    #picture = "Bandits"
  
	trigger = {
	
		
	
		civilized = yes
		any_owned_province = {
			is_colonial = yes
			OR = {
				AND = {
					colonial_policy = settlement
					OR = {
						has_province_modifier = exploitation
						has_province_modifier = self_rule
						NOT = { has_province_modifier = settler_colonization }
					}
				}
				AND = {
					colonial_policy = exploitation
					OR = {
						has_province_modifier = settlement
						has_province_modifier = self_rule
						NOT = { has_province_modifier = exploitation }
					}
				}
				AND = {
					colonial_policy = self_rule
					OR = {
						has_province_modifier = exploitation
						has_province_modifier = settlement
						NOT = { has_province_modifier = self_rule }
					}
				}
				AND = {
					colonial_policy = no_colonial_policy
					OR = {
						has_province_modifier = exploitation
						has_province_modifier = settlement
						has_province_modifier = self_rule
					}
				}
			}
		}
	}
				
	mean_time_to_happen = {
        months = 1
	}

    option = {
        name = "EVT500116PTA" 
		any_owned = {
			limit = {
				is_colonial = yes
				#NOT = { has_province_modifier = self_rule }
			}
			remove_province_modifier = exploitation
			remove_province_modifier = self_rule
			remove_province_modifier = settlement
		}
		if = {
			limit = {
				THIS = {
					colonial_policy = exploitation
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = exploitation }
					}
					add_province_modifier = {
						name = exploitation
						duration = 36525
					}
				}
			}
		}
		if = {
			limit = {
				THIS = {
					colonial_policy = settlement
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = settler_colonization }
					}
					add_province_modifier = {
						name = settler_colonization
						duration = 36525
					}
				}
			}
		}
		if = {
			limit = {
				THIS = {
					colonial_policy = self_rule
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = self_rule }
					}
					add_province_modifier = {
						name = self_rule
						duration = 36525
					}
				}
			}
		}
		
    }
}
