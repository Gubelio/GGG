country_event = {
    id = 50115
    title = "EVT50115NAME"
    desc = "EVT50115DESC"
    #picture = "Bandits"
  
	trigger = {
		civilized = yes
		NOT = { colonial_policy = no_colonial_policy }
		any_owned_province = {
			OR = {
				AND = {
					NOT = { has_province_modifier = exploitation }
					is_colonial = yes
				}
				AND = {
					NOT = { has_province_modifier = settler_colonization }
					is_colonial = yes
				}
				AND = {
					NOT = { has_province_modifier = self_rule }
					is_colonial = yes
				}
			}
		}
	}
				
	mean_time_to_happen = {
        months = 1
	}

    option = {
        name = "EVT500115PTA" 
		if = {
			limit = {
				THIS = {
					colonial_policy = exploitation
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = exploitation }
					}
					add_province_modifier = {
						name = exploitation
						duration = 36525
					}
				}
			}
		}
		if = {
			limit = {
				THIS = {
					colonial_policy = settlement
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = settler_colonization }
					}
					add_province_modifier = {
						name = settler_colonization
						duration = 36525
					}
				}
			}
		}
		if = {
			limit = {
				THIS = {
					colonial_policy = self_rule
				}
			}
			THIS = {
				any_owned = {
					limit = {
						is_colonial = yes
						NOT = { has_province_modifier = self_rule }
					}
					add_province_modifier = {
						name = self_rule
						duration = 36525
					}
				}
			}
		}
    }
}
