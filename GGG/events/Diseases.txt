			NOT = {

			}
		
		}
	
	}
	
	mean_time_to_happen = {
		months = 300
	
	}
	
	option = {
		name = "Quaratine the region!"
		treasury = -100000
		
		random_owned = {
			limit = {
				OR = {						# THESE ARE EXAMPLES! GROUP POSSIBLE ELIGIBLE PROVINCE CONDITIONS INTO AND GROUPS
					AND = {
						terrain = plains
						has_pop_type = farmers
					
					}

						terrain = hills
						has_pop_type = serfs
					}
				
				}
			
			}
