country_event = {
	id = 50109
	title = "EVTNAME50109" #locust plague incoming
	desc = "EVTDESC50109"
	picture = "locust"
	
    trigger = {
		any_owned_province = {
			OR = {
				terrain = plains
				terrain = savanna
				terrain = farmlands
			}
		}
    }
	
	mean_time_to_happen = {
        months = 150
        modifier = {
            factor = 1.5
			NOT = {
				OR = {
					month = july #First Rain
					month = august #High Monsoon
					month = september #Late Monsoon
					month = october #Harvesttime
				}
			}
        }
        modifier = {
            factor = 0.8
			month = july #First Rain
        }
        modifier = {
            factor = 0.9
			month = august #High Monsoon
        }
        modifier = {
            factor = 0.4
			any_owned_province = {
				has_province_modifier = locust_plague
			}
        }

	}
	
	option = {
		name = "EVT50109OPTA"
		random_state = {
			limit = {
				any_owned_province = {
					OR = {
						terrain = savanna
						terrain = plains
						terrain = farmlands
					}
				}
			}
			any_owned = {
				limit = {
					OR = {
						terrain = savanna
						terrain = plains
						terrain = farmlands
					}
				}
				add_province_modifier = { name = locust_threat duration = 365 }
			}
		}
		set_country_flag = locust_plague_incoming
	}
}
