	id = 50008 
	title = "EVTNAME50008"
	desc = "EVTDESC50008"
	#picture = "Socialism"
	
	trigger = {
		civilized = yes
		scriptural_law = morality_police 
		clerical_law = empowered_clergy

		    legislative_system = appointed_legislatures

		NOT = { government = theocracy }
	}

	mean_time_to_happen = {
		months = 1
	}

	option = {
		name = "EVT50008OPTA"
		government = theocracy
	}
}
