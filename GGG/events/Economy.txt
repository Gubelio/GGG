country_event = { 
    id = 50084 
    title = "EVTNAME50084" #developing economy
    desc = "EVTDESC50084"
	#picture = "NiizMoji"

    trigger = {
		industrial_score = 25
		civilized = yes

		NOT = {
				has_country_modifier = developing_economy
				has_country_modifier = industrialising_economy
				has_country_modifier = industrial_economy
			}
		}
	

    option = {
        name = "EVT50084OPTA"
		add_country_modifier = { name = developing_economy duration = -1 }
	}
}
