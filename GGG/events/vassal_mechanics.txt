country_event = { #tribute event
	id = 50358
	title = "EVTNAME50358"
	desc = "EVTDESC50358"
	#picture = "x"
	
	trigger = {
		has_country_flag = tributary_country
		has_country_flag = bleed_dry_vassal
		is_vassal = yes
		
		OR = {
			AND = {
				ai = yes
				money = 20000
			}
			
			AND = {
				ai = no
				money = 100000
			
			}
		
		
		}
		
	}
	
	mean_time_to_happen = {
		months = 4
	
	}
	
	option = {
		name = "This is our lot, it seems."
		
		prestige = -5
		
		any_pop = {
			militancy = 0.5
			consciousness = 0.5
		}
		
		if = {
			limit = {
				THIS = {
					ai = yes
				
				}
			}
			any_country = {
				limit = {
					is_our_vassal = THIS
				}
			
				change_variable = {
					which = tribute_recieved
					value = 18
				}
			
			}
			THIS = {
				treasury = -18000
			
			}
		
		}
		
		if = {
			limit = {
				THIS = {
					ai = no
				
				}
			}
			any_country = {
				limit = {
					is_our_vassal = THIS
				}
			
				change_variable = {
					which = tribute_recieved
					value = 90
				}
			
			}
			THIS = {
				treasury = -90000
			
			}
		
		}
	
		
	
	}
}

country_event = {
	id = 50362 
	title = "EVTNAME50362" 
	desc = "EVTDESC50362"
	picture = "chinese_palace_event"
	
	is_triggered_only = yes

    option = {
        name = "It is time to do away with the pretenses of vassalhood."
		clr_country_flag = negotiating_with_vassals
        if = {
            limit = {
                FROM = {
                    NOT = { 
                        has_country_modifier = new_overlordship
			has_country_modifier = colonial_overlordship
                    }
                    primary_culture = THIS
                }
            }

            THIS = {
                inherit = FROM
                set_country_flag = vass_ann_suc
            }

        }

        if = {
            limit = {
                THIS = {
                    NOT = { 
                        has_country_flag = vass_ann_suc
                    }
                }
            }
            THIS = {
                release_vassal = FROM
            }
                
            FROM = {
			    add_casus_belli = {
				    target = THIS
				    type = make_puppet
				    months = 12
			    }
            }
            THIS = {
			    war = {
				    target = FROM
				    attacker_goal = { casus_belli = make_puppet }
				    defender_goal = { casus_belli = status_quo }
                }
            }
        }
        clr_country_flag = vassal_ann_suc

    option = {

					name = national_resistance
					duration = 200
				}

			    war = {
				    target = FROM

				    defender_goal = { casus_belli = status_quo }
                }
            }
        }

        clr_country_flag = vass_int_suc
		


    }

    


    option = {
        name = "We no longer with to be their overlords. Set them free."
		clr_country_flag = negotiating_with_vassals

        prestige = -10
        infamy = -0.5

        FROM = {
            remove_country_modifier = foreign_rule
            remove_country_modifier = new_overlordship
	    remove_country_modifier = colonial_overlordship

        }

    }

	
	option = {
		name = "We wish to raise their tribute: bleed them dry!"

			set_country_flag = bleed_dry_vassal
