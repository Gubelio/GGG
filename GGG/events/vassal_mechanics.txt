country_event = { #tribute event
	id = 50358
	title = "EVTNAME50358"
	desc = "EVTDESC50358"
	#picture = "x"
	
	trigger = {
		has_country_flag = tributary_country
		has_country_flag = bleed_dry_vassal
		is_vassal = yes
		
		OR = {
			AND = {
				ai = yes
				money = 20000
			}
			
			AND = {
				ai = no
				money = 100000
			
			}
		
		
		}
		
	}
	
	mean_time_to_happen = {
		months = 4
	
	}
	
	option = {
		name = "This is our lot, it seems."
		
		prestige = -5
		
		any_pop = {
			militancy = 2
			consciousness = 2
		}
		
		if = {
			limit = {
				THIS = {
					ai = yes
				
				}
			}
			any_country = {
				limit = {
					is_our_vassal = THIS
				}
			
				change_variable = {
					which = tribute_recieved
					value = 18
				}
			
			}
			THIS = {
				treasury = -18000
			
			}
		
		}
		
		if = {
			limit = {
				THIS = {
					ai = no
				
				}
			}
			any_country = {
				limit = {
					is_our_vassal = THIS
				}
			
				change_variable = {
					which = tribute_recieved
					value = 90
				}
			
			}
			THIS = {
				treasury = -90000
			
			}
		
		}
	
		
	
	}
}
