

country_event = { 
    id = 50066 
    title = "EVT50066" #Treaty port of Niiz Moji
    desc = "EVTDESC50066"
	picture = "NiizMoji"

    trigger = {
		owns = 3055
		civilized = yes
		OR = {
			capital_scope = { continent = north_america }
			tag = AYU
		}
		NOT = {
			3055 = {
				has_province_modifier = treaty_port
			}
		}
    }


    option = {
        name = "EVT50066OPTA"
		3055 = {
			add_province_modifier = { name = treaty_port duration = 36525 }
		}
	}
}
