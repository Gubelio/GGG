powerful_barons = {
	trigger = {
		tag = TAK
		check_variable = {
			which = barons_influence
			value = 3
		}
		NOT = {
			has_country_flag = dominant_clergy
			has_country_flag = dominant_barons
			has_country_flag = dominant_assembly
		}
	}
	
	rich_vote = 0.1
	naval_organisation = 0.05
	naval_unit_start_experience = 10
	supply_consumption = 0.05
	administrative_efficiency_modifier = -0.03
	
}

international_sanctions = {
        trigger = {
		NOT = {
			war_exhaustion = 45
		}
		badboy = 1.3 # 46.8
		NOT = { has_global_flag = sanctions_disabled }
		NOT = { 
			check_variable = {
				which = international_hegemon
				value = 4
			}
		}
	}

    import_cost = 0.5
    diplomatic_points_modifier = -0.6
    non_accepted_pop_militancy_modifier = 0.2
    prestige = -0.3
    war_exhaustion = 0.5
    core_pop_militancy_modifier = 0.1
    icon = 6
}
